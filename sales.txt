import pandas as pd
import matplotlib.pyplot as plt

from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans, AgglomerativeClustering
from scipy.cluster.hierarchy import dendrogram, linkage


df = pd.read_csv("sales_data_sample.csv", encoding="latin1")

df


data = df[["QUANTITYORDERED", "PRICEEACH", "SALES", "MSRP"]]
print(data.describe())


scaler = StandardScaler()
data_scaled = scaler.fit_transform(data)


wcss = []
for k in range(1, 11):
    model = KMeans(n_clusters=k, random_state=42, n_init=10)
    model.fit(data_scaled)
    wcss.append(model.inertia_)
    print("k =", k, " WCSS =", round(model.inertia_, 2))

plt.plot(range(1, 11), wcss, marker="o")
plt.xlabel("number of clusters")
plt.ylabel("WCSS")
plt.title("Elbow method")
plt.show()


kmeans = KMeans(n_clusters=3, random_state=42, n_init=10)
kmeans.fit(data_scaled)
labels = kmeans.labels_
counts = []
for c in range(3):
    counts.append(len(labels[labels == c]))


plt.scatter(data["QUANTITYORDERED"], data["SALES"], c=labels)
plt.xlabel("quantity ordered")
plt.ylabel("sales")
plt.title("K-Means clusters")
plt.show()


sample = data_scaled[:200]

linked = linkage(sample, method="ward")

plt.figure(figsize=(10, 5))
dendrogram(linked)
plt.title("Dendrogram")
plt.xlabel("points")
plt.ylabel("distance")
plt.show()


hier = AgglomerativeClustering(n_clusters=3, linkage="ward")
hier_labels = hier.fit_predict(data_scaled)

print("points in each cluster:", [len(hier_labels[hier_labels == c]) for c in range(3)])

plt.scatter(data["QUANTITYORDERED"], data["SALES"], c=hier_labels)
plt.xlabel("quantity ordered")
plt.ylabel("sales")
plt.title("Hierarchical clusters")
plt.show()
